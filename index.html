function generateCode() {

  const chars = 'ABCDEFGHJKLMNPQRSTUVWXYZ23456789';

  let code = 'FARM-';

  for (let i = 0; i < 4; i++) code += chars[Math.floor(Math.random() * chars.length)];

  return code;

}

function regenerateCode() {

  pendingCode = generateCode();

}


async function createNewFarm() {

  if (!pendingCode) { showToast('กดสร้างรหัสก่อน'); return; }

  setSyncStatus('syncing');

  try {

    const batch = db.batch();

    DEFAULT_MEDS.forEach(m => {

      batch.set(db.doc(`farms/${pendingCode}/meds/${m.id}`), m);

    });

    batch.set(db.doc(`farms/${pendingCode}/config/settings`), {

      trees: DEFAULT_TREES,

      createdAt: firebase.firestore.FieldValue.serverTimestamp()

    });

    await batch.commit();

    localStorage.setItem('durianFarmCode', pendingCode);

    farmCode = pendingCode;

    hideFarmScreen();

    updateFarmCodeDisplay();

    startListeners();

  } catch(e) {

    setSyncStatus('offline');

    showToast('เกิดข้อผิดพลาด กรุณาตรวจสอบอินเทอร์เน็ต');

  }

}


async function joinFarm() {

  const code = document.getElementById('joinCodeInput').value.trim().toUpperCase();

  if (!code) { showToast('กรุณาใส่รหัสฟาร์ม'); return; }

  document.getElementById('joinError').style.display = 'none';

  setSyncStatus('syncing');

  try {

    const settingsDoc = await db.doc(`farms/${code}/config/settings`).get();

    if (!settingsDoc.exists) {

      document.getElementById('joinError').style.display = 'block';

      setSyncStatus('offline'); return;

    }

    localStorage.setItem('durianFarmCode', code);

    farmCode = code;

    hideFarmScreen();

    updateFarmCodeDisplay();

    startListeners();

  } catch(e) {

    document.getElementById('joinError').style.display = 'block';

    setSyncStatus('offline');

  }

}

function startListeners() {

  listeners.forEach(u => u()); listeners = [];

  const fp = `farms/${farmCode}`;


  const u1 = db.collection(`${fp}/records`).orderBy('date','desc').onSnapshot(snap => {

    cache.records = snap.docs.map(d => ({...d.data(), id: d.id}));

    rerenderCurrent();

    setSyncStatus('connected');

  }, () => setSyncStatus('offline'));


  const u2 = db.collection(`${fp}/meds`).onSnapshot(snap => {

    cache.meds = snap.docs.map(d => ({...d.data(), id: d.id}));

    rerenderCurrent();

  }, () => {});


  const u3 = db.doc(`${fp}/config/settings`).onSnapshot(doc => {

    if (doc.exists) cache.trees = doc.data().trees || DEFAULT_TREES;

    updateTreeSelect(); rerenderCurrent();

  }, () => {});


  listeners = [u1, u2, u3];

}

}

function fmtDate(s) { if(!s)return'-'; const d=new Date(s); return d.toLocaleDateString('th-TH',{day:'2-digit',month:'short',year:'numeric'}); }

function fmtDT(d,t){ return t ? fmtDate(d)+' '+t+' น.' : fmtDate(d); }

function daysFrom(s){ if(!s)return null; const t=new Date();t.setHours(0,0,0,0);const d=new Date(s);d.setHours(0,0,0,0);return Math.round((d-t)/86400000); }

function stockStatus(m) {

  if(!m.total && !m.current) return 'none';

  if(m.current<=0) return 'critical';

  if(m.minAlert>0 && m.current<=m.minAlert) return 'low';

  return 'ok';

}

function stockLabel(m) {

  const st=stockStatus(m);

  if(st==='none') return '<span style="color:#90a4ae;font-size:.75rem">ยังไม่ตั้งค่า</span>';

  const pct=m.total>0?Math.min(100,Math.round(m.current/m.total*100)):100;

  const cls=st==='ok'?'ok':st==='low'?'low':'crit';

  const lbl=st==='ok'?'เพียงพอ':st==='low'?'เหลือน้อย':'วิกฤต!';

  return `<div class="sbw"><div class="sbar"><div class="sfill ${cls}" style="width:${pct}%"></div></div><span class="stxt ${cls}">${m.current}${m.unit}<br><small>${lbl}</small></span></div>`;

}

  const blob = new Blob([content],{type:'text/csv;charset=utf-8'});

  const a = Object.assign(document.createElement('a'),{href:URL.createObjectURL(blob),download:filename});

  a.click(); URL.revokeObjectURL(a.href);

}

    const record={id:uid(),tree,date,time,medicine,amount,method,nextDate,notes,createdAt:Date.now()};

function renderHistory() {

  const search=(document.getElementById('searchInput').value||'').toLowerCase();

  const fMonth=document.getElementById('filterMonth').value;

  ftEl.innerHTML='<option value="">ทั้งหมด</option>'+trees.map(t=>`<option value="${t}" ${t===cur?'selected':''}>${t}</option>`).join('');

  let f=records.filter(r=>{

    if(fTree&&r.tree!==fTree)return false;

    if(fMonth&&(!r.date||r.date.slice(0,7)!==fMonth))return false;

    if(search&&![r.tree,r.medicine,r.amount,r.notes,r.method].join(' ').toLowerCase().includes(search))return false;

    return true;

  });

  const tbody=document.getElementById('historyBody');

  if(!f.length){tbody.innerHTML='<tr><td colspan="9" class="empty"><span class="ei">🔍</span>ไม่พบข้อมูล</td></tr>';return;}

  tbody.innerHTML=f.map((r,i)=>{

    const days=r.nextDate?daysFrom(r.nextDate):null;

    const nc=r.nextDate?`<span style="color:${days<0?'var(--r)':days<=3?'var(--o)':'var(--gd)'}">${fmtDate(r.nextDate)}${days<0?' 🚨':days===0?' ⚠️':''}</span>`:'-';

    return`<tr><td>${f.length-i}</td><td style="white-space:nowrap">${fmtDT(r.date,r.time)}</td><td><strong>🌳${r.tree}</strong></td><td>${r.medicine}</td><td>${r.amount||'-'}</td><td>${r.method||'-'}</td><td style="white-space:nowrap">${nc}</td><td style="max-width:120px;font-size:.76rem;color:var(--gr)">${r.notes||'-'}</td><td><button class="btn btn-d btn-xs" onclick="deleteRecord('${r.id}')">🗑️</button></td></tr>`;

  }).join('');

}
